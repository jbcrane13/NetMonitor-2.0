func checkpoint(_ url: URL) {
    let p = Process(); p.executableURL = URL(fileURLWithPath: "/usr/bin/sqlite3")
    p.arguments = [url.path, "PRAGMA wal_checkpoint(TRUNCATE); PRAGMA journal_mode=DELETE;"]; try! p.run(); p.waitUntilExit()
}
